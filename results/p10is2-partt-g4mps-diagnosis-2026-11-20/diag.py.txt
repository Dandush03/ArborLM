def names_of(arm):  # type: ignore[no-untyped-def]
    ids = {}
    for attr in ("store", "adapter", "module"):
        m = getattr(arm, attr, None)
        if m is not None and hasattr(m, "named_parameters"):
            ids.update({id(p): f"{attr}.{n}" for n, p in m.named_parameters()})
    return [ids.get(id(p), f"param{i}") for i, p in enumerate(arm.trainable())]


def run(dev, hist=False, dtype=None):  # type: ignore[no-untyped-def]
    if dtype is not None:
        torch.set_default_dtype(dtype)
        w.student.to(dtype)
    arm, rows, opts = engine.build_cell_arm(w, spec, None)
    res["lr"] = float(opts["rate"])
    names = names_of(arm)
    init = [p.detach().cpu().clone().double() for p in arm.trainable()]
    keep, w.device = w.device, dev
    snaps: list = []
    kw = {"snapshots": list(range(1, 21)), "on_snapshot": lambda u, a: snaps.append([p.detach().cpu().clone().double() for p in a.trainable()])} if hist else {}
    with warnings.catch_warnings(record=True) as caught:
        warnings.simplefilter("always")
        try:
            rec = engine.train_cell(w, spec, arm, rows, opts, 20, record_losses=20, **kw)
        finally:
            w.device = keep
            if dtype is not None:
                torch.set_default_dtype(torch.float32)
                w.student.to(torch.float32)
    mps_warn = sorted({str(c.message)[:160] for c in caught if "MPS" in str(c.message) or "mps" in str(c.message)})
    assert rec["status"] == "COMPLETE", rec["error"]
    return {"losses": rec["losses_head"], "final": [p.detach().cpu().clone().double() for p in arm.trainable()], "init": init, "snaps": snaps, "names": names, "warn": mps_warn}


def rel(a, b):  # type: ignore[no-untyped-def]
    return [float((x - y).norm() / y.norm().clamp_min(1e-30)) for x, y in zip(a, b, strict=True)]


if part == "A":
    t = time.time()
    cpu = run(torch.device("cpu"), hist=True)
    res["cpu_seconds_20_updates"] = time.time() - t
    t = time.time()
    mps = run(w.device, hist=True)

    rows = []
    for i, n in enumerate(cpu["names"]):
        pc, pm, p0 = cpu["final"][i], mps["final"][i], cpu["init"][i]
        diff, nc, ch = float((pm - pc).norm()), float(pc.norm()), float((pc - p0).norm())
        rows.append({"i": i, "name": n, "shape": list(pc.shape), "numel": pc.numel(), "norm_cpu": nc, "norm_init": float(p0.norm()), "diff_mps_cpu": diff, "change_cpu": ch, "rel_to_final": diff / max(nc, 1e-30), "rel_to_change": diff / max(ch, 1e-30)})
    res["tensors"] = rows
    # 2. growth per update and per-update loss differences
    growth = []
    for u in range(20):
        r = rel(mps["snaps"][u], cpu["snaps"][u])
        worst = max(range(len(r)), key=lambda j: r[j])
        growth.append({"update": u + 1, "max_rel": max(r), "worst_tensor": cpu["names"][worst], "loss_cpu": cpu["losses"][u], "loss_mps": mps["losses"][u], "loss_diff": abs(cpu["losses"][u] - mps["losses"][u])})
    res["growth"] = growth
    res["gate_metric_reproduced"] = max(rel(mps["final"], cpu["final"]))
    # elementwise structure of the worst tensor
    worst = max(rows, key=lambda r: r["rel_to_final"])["i"]
    pc, pm, p0 = cpu["final"][worst], mps["final"][worst], cpu["init"][worst]
    d = (pm - pc).abs().flatten()
    chg = (pc - p0).abs().flatten()
    lr = res["lr"]
    sat = chg > 0.5 * lr * 20
    res["worst_tensor_elements"] = {"name": cpu["names"][worst], "numel": int(d.numel()), "lr": lr, "share_of_sq_diff_in_top_1pct_elements": float((d.sort(descending=True).values[: max(1, d.numel() // 100)] ** 2).sum() / (d**2).sum().clamp_min(1e-60)), "fraction_with_change_below_1e-9": float((chg < 1e-9).double().mean()), "fraction_elements_moved_over_half_lr_times_updates": float(sat.double().mean()), "mean_abs_diff": float(d.mean()), "max_abs_diff": float(d.max())}
    # 3. controls

    res["control_cpu_cpu_loss_diff"] = max(abs(a - b) for a, b in zip(cpu2["losses"], cpu["losses"], strict=True))
    mps2 = run(w.device)

    res["control_mps_mps_per_tensor"] = rel(mps2["final"], mps["final"])

        res["control_cpu_f32_vs_f64_per_tensor"] = rr
        res["control_cpu_f32_vs_f64_loss_diff"] = max(abs(a - b) for a, b in zip(f64["losses"], cpu["losses"], strict=True))
        res["control_mps_vs_f64_max_rel"] = max(rel(mps["final"], f64["final"]))
    except Exception as e:  # noqa: BLE001
        res["control_f64_error"] = f"{type(e).__name__}: {e}"[:300]
    json.dump(res, open(OUT / "partA.json", "w"), indent=1)
    print("part A written", flush=True)
    # 4. first-step gradients on the same first batch, CPU against MPS
    plan_s = w.plan(0)

    grads = {}
    for name, dev in (("cpu", torch.device("cpu")), ("mps", w.device)):
        arm, _r, _o = engine.build_cell_arm(w, spec, None)

    res["first_step_gradients"] = {"loss_cpu": grads["cpu"][0], "loss_mps": grads["mps"][0], "tensors": gl}
