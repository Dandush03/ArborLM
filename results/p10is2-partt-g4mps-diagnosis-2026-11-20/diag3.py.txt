def names_of(arm):  # type: ignore[no-untyped-def]
    ids = {}
    for attr in ("store", "adapter", "module"):
        m = getattr(arm, attr, None)
        if m is not None and hasattr(m, "named_parameters"):
            ids.update({id(p): f"{attr}.{n}" for n, p in m.named_parameters()})
    return [ids.get(id(p), f"param{i}") for i, p in enumerate(arm.trainable())]


def run(dev, hist=False, dtype=None):  # type: ignore[no-untyped-def]
    if dtype is not None:
        torch.set_default_dtype(dtype)
        w.student.to(dtype)
    arm, rows, opts = engine.build_cell_arm(w, spec, None)
    res["lr"] = float(opts["rate"])
    names = names_of(arm)
    init = [p.detach().cpu().clone().double() for p in arm.trainable()]
    keep, w.device = w.device, dev
    snaps: list = []
    kw = {"snapshots": list(range(1, 21)), "on_snapshot": lambda u, a: snaps.append([p.detach().cpu().clone().double() for p in a.trainable()])} if hist else {}
    with warnings.catch_warnings(record=True) as caught:
        warnings.simplefilter("always")
        try:
            rec = engine.train_cell(w, spec, arm, rows, opts, 20, record_losses=20, **kw)
        finally:
            w.device = keep
            if dtype is not None:
                torch.set_default_dtype(torch.float32)
                w.student.to(torch.float32)
    mps_warn = sorted({str(c.message)[:160] for c in caught if "MPS" in str(c.message) or "mps" in str(c.message)})
    assert rec["status"] == "COMPLETE", rec["error"]
    return {"losses": rec["losses_head"], "final": [p.detach().cpu().clone().double() for p in arm.trainable()], "init": init, "snaps": snaps, "names": names, "warn": mps_warn}


def rel(a, b):  # type: ignore[no-untyped-def]
    return [float((x - y).norm() / y.norm().clamp_min(1e-30)) for x, y in zip(a, b, strict=True)]
