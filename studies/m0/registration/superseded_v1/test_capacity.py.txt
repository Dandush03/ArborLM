# ================================================================================ made-up lineages


def lin(lid: str, build: Callable[[int, random.Random], Built]) -> Lineage:
    return Lineage(lid, lid.split(".")[0], "toy", build)


def rand_toy(lid: str, m: int) -> Lineage:
    """Items 0..m-1 drawn uniformly: exactly m distinct items, a pure function of the stream."""

    def build(level: int, rng: random.Random) -> Built:
        i = rng.randrange(m)
        return Built(f"{lid} level {level} item {i}.", {"i": i}, str(i), "number", None)

    return lin(lid, build)


def fresh_toy(lid: str, dup: float) -> Lineage:
    """With probability ``dup`` the item is item 0; otherwise a new 48-bit item: duplicate-heavy, a pure function of the stream."""

    def build(level: int, rng: random.Random) -> Built:
        i = 0 if rng.random() < dup else rng.getrandbits(48) + 1
        return Built(f"{lid} L{level} #{i}", {"i": i}, str(i), "number", None)

    return lin(lid, build)


_TICKS: weakref.WeakKeyDictionary[random.Random, int] = weakref.WeakKeyDictionary()


def _tick(rng: random.Random) -> int:
    n = _TICKS.get(rng, 0) + 1
    _TICKS[rng] = n
    return n


def period_toy(lid: str, period: int) -> Lineage:
    """Stateful: a new item at attempts 1, 1 + p, 1 + 2p, ...; every other attempt repeats item 0 (no cache with this one)."""

    def build(level: int, rng: random.Random) -> Built:
        n = _tick(rng)
        i = (n - 1) // period + 1 if (n - 1) % period == 0 else 0
        return Built(f"{lid} L{level} #{i}", {"i": i}, str(i), "number", None)

    return lin(lid, build)


def give_up_toy(lid: str, after: int) -> Lineage:
    """Stateful: ``after`` distinct items, then every call asks for a retry, so build_item gives up after 500."""

    def build(level: int, rng: random.Random) -> Built:
        n = _tick(rng)
        if n > after:
            raise Retry
        return Built(f"{lid} L{level} #{n}", {"i": n}, str(n), "number", None)

    return lin(lid, build)


def names_toy(lid: str, m: int) -> Lineage:
    """A person's name (about 130 choices) and a number below m: m distinct items once names are relabelled."""

    def build(level: int, rng: random.Random) -> Built:
        n, k = rng.choice(NAMES), rng.randrange(m)
        return Built(f"{n} has {k} apples.", {"name": n, "k": k}, str(k), "number", None)

    return lin(lid, build)


class ToySource:
    def __init__(self, lineages: dict[str, Lineage], solvers: dict[str, Callable[[str, Any], Any]] | None = None) -> None:
        self.lins = lineages
        self.solvers = solvers or {}

    def lineage(self, lid: str) -> Lineage:
        return self.lins[lid]

    def solver(self, lid: str) -> Callable[[str, Any], Any]:
        return self.solvers[lid]


def agree_solver(question: str, structured: Any) -> str:
    return str(structured["i"])


def reference(lineage: Lineage, level: int, target: int, key_fn: Callable[..., bytes] = dk.item_key) -> tuple[int, str, int]:
    """The text's "how a count is made", written again and independently of ``capacity.count``: (count, ended, attempts)."""
    rng = stream(400002, "capacity", lineage.eval, lineage.id, str(level))
    keys: set[bytes] = set()
    attempts = 0
    while len(keys) < target:
        attempts += 1
        if attempts > 40 * target + 1000:
            return len(keys), "budget", attempts - 1
        try:
            it = build_item(lineage, level, rng, "-", "x")
        except RuntimeError:
            return len(keys), "gave_up", attempts
        keys.add(key_fn(lineage.id, it.question, it.structured))
    return len(keys), "target", attempts


def n_keys(d: cap.Draw) -> int:
    return len({d.keys[i : i + 16] for i in range(0, len(d.keys), 16)})

        for k in range(1, 12):

def test_slack_tables_of_the_text() -> None:
    for k, row in SLACK_TEXT.items():
        assert [k * cap.demand_lin(g, k) - cap.demand(g) for g in range(1, 20)] == row
    zero = {k: [g for g in range(1, 20) if k * cap.demand_lin(g, k) == cap.demand(g)] for k in range(2, 9)}
    assert zero[2] == [1, 2, 4, 5, 7, 8, 10, 16]
    assert zero[4] == [1, 2, 4, 5, 8, 10]
    assert zero[5] == [1, 2, 4, 5, 8, 10, 16]
    assert zero[3] == zero[6] == zero[7] == zero[8] == []
    cells = {(g, k): k * cap.demand_lin(g, k) - cap.demand(g) for g in range(1, 20) for k in range(2, 9)}
    assert max(cells.values()) == 160 and [gk for gk, v in cells.items() if v == 160] == [(16, 8)]
    assert min(cells.values()) == 0

def key(text: str, lid: str = "R1.toy") -> bytes:
    return dk.item_key(lid, text, {})


def test_names_relabelled_same_key() -> None:
    assert key("Ada gave Ben 3 apples.") == key("Cara gave Dev 3 apples.")
    assert key("Ada gave Ben 3 apples.") == key("Ben gave Ada 3 apples.")  # a one-to-one renaming
    assert key("Ada's teapots cost 3.") == key("Ben's drums cost 3.")  # possessive and plural stay in the text
    assert key("Ada has an apple.") == key("Ben has a book.")  # the article agrees with the name
    assert key("Arden is near Brook.") == key("Calder is near Dunmore.")  # places
    assert key("The concert came before the book sale.") == key("The film night came before the art auction.")  # events
    assert key("Sit with a kokex and a zazin.") == key("Sit with a banak and a dedox.")  # invented words
    assert key("A yo-yo and 3 yo-yos.") == key("A kettle and 3 kettles.")  # hyphenated entry with a plural


def test_exact_duplicates_are_one_key() -> None:
    assert key("Ada has 3 apples.") == key("Ada has 3 apples.")


def test_anything_else_is_a_different_key() -> None:
    base = key("Ada gave Ben 3 apples.")
    for other in [
        "Ada gave Ben 4 apples.",  # a number
        "Ada gave Ben 3 oranges.",  # "oranges" is a label word here, not a name
        "Ben gave 3 apples to Ada.",  # order of presentation
        "Ada gave Ada 3 apples.",  # the pattern of repeated names
        "Ada gave Ben 3 apples",  # punctuation
        "Ada gave  Ben 3 apples.",  # whitespace is not normalised
        "ada gave Ben 3 apples.",  # lower case is not the name
        "Ada gives Ben 3 apples.",  # another word
        "Ada gave Ben 3 apples. ",
    ]:
        assert key(other) != base, other
    assert key("A red kettle.") != key("A blue kettle.")  # labels are not relabelled (reading (ii) would)
    assert key("An orange.") != key("A book.")  # "orange" is in the object list and the colour list: a label
    assert key("Letter b follows a.") != key("Letter c follows a.")  # letters
    assert key("It was Friday.") != key("It was Monday.")  # weekdays


def test_a_name_inside_a_longer_word_is_not_a_name() -> None:
    assert key("Adam is here.") != key("Cara is here.")
    assert key("Bookshelf") != key("Drumshelf")


def test_one_counter_numbers_names_in_order_of_first_appearance() -> None:

def _c1(program: list[list]) -> tuple[str, dict[str, Any]]:
    q = f"Here is a Python program.\n\n{render(program)}\n\nWhat exactly does it print? It prints a single line."
    return q, {"program": json.loads(json.dumps(program))}


def c1key(program: list[list]) -> bytes:
    q, st = _c1(program)
    return dk.item_key("C1.toy", q, st)


def _prog(a: str, b: str, lit: int = 5, op: str = "+") -> list[list]:
    return [["set", a, ["n", lit]], ["set", b, ["b", op, ["v", a], ["n", 2]]], ["print", ["call", "len", ["list", ["v", a], ["v", b]]]]]


def test_c1_identifiers_relabelled() -> None:
    assert c1key(_prog("a", "b")) == c1key(_prog("total", "res"))
    assert c1key(_prog("a", "b")) == c1key(_prog("b", "a"))  # a swap is a renaming too
    assert c1key(_prog("a", "b", lit=5)) != c1key(_prog("a", "b", lit=6))  # a literal
    assert c1key(_prog("a", "b", op="+")) != c1key(_prog("a", "b", op="-"))  # an operator
    assert c1key(_prog("a", "a")) != c1key(_prog("a", "b"))  # the pattern of reused names
    other = [["set", "a", ["n", 5]], ["print", ["v", "a"]]]
    assert c1key(other) != c1key(_prog("a", "b"))


def test_c1_builtins_methods_strings_and_layout() -> None:
    p1 = [["set", "s", ["s", "ab"]], ["print", ["m", ["v", "s"], "upper"]]]
    p2 = [["set", "t", ["s", "ab"]], ["print", ["m", ["v", "t"], "upper"]]]
    p3 = [["set", "s", ["s", "ab"]], ["print", ["m", ["v", "s"], "lower"]]]
    assert c1key(p1) == c1key(p2)
    assert c1key(p1) != c1key(p3)  # a method name is not a variable name
    q, st = _c1(p1)
    assert "upper" in dk.key_text("C1.toy", q, st) and "print" in dk.key_text("C1.toy", q, st)
    s1 = [["set", "a", ["n", 1]], ["print", ["s", "a"]]]
    s2 = [["set", "b", ["n", 1]], ["print", ["s", "a"]]]
    s3 = [["set", "b", ["n", 1]], ["print", ["s", "b"]]]
    assert c1key(s1) == c1key(s2) and c1key(s2) != c1key(s3)  # letters inside a string literal are not names
    f1 = [["def", "f", ["x"], [["ret", ["b", "+", ["v", "x"], ["n", 1]]]]], ["print", ["call", "f", ["n", 2]]]]
    f2 = [["def", "calc", ["q"], [["ret", ["b", "+", ["v", "q"], ["n", 1]]]]], ["print", ["call", "calc", ["n", 2]]]]
    assert c1key(f1) == c1key(f2)  # function and parameter names
    deeper = [["set", "a", ["n", 1]], ["if", ["c", ">", ["v", "a"], ["n", 0]], [["print", ["v", "a"]]], []]]
    flat = [["set", "a", ["n", 1]], ["print", ["v", "a"]]]
    assert c1key(deeper) != c1key(flat)

def test_attempt_budget_binding_and_its_boundary() -> None:
    """period_toy(p): new items "#1", "#2", ... at attempts 1, 1 + p, 1 + 2p, ...; every other attempt is item "#0", a key too.
    With T = 100 the budget is 40 * 100 + 1,000 = 5,000 attempts; the 100th key is #0 plus the 99th new item, at attempt 98 p + 1."""
    on = cap.count(period_toy("R1.p51", 51), 0, 100)  # 98 * 51 + 1 = 4,999 <= 5,000: reached, one attempt before the budget
    assert (on.count, on.ended, on.attempts) == (100, "target", 4999)
    last = cap.count(period_toy("R1.p51b", 51), 0, 100)
    assert last.attempts <= 5000 and n_keys(last) == 100
    over = cap.count(period_toy("R1.p52", 52), 0, 100)  # 98 * 52 + 1 = 5,097 > 5,000: the budget binds at 97 new items and #0
    assert (over.count, over.ended, over.attempts) == (98, "budget", 5000)
    exact = cap.count(period_toy("R1.p5000", 5000), 0, 1)
    assert (exact.count, exact.ended, exact.attempts) == (1, "target", 1)
    just = cap.count(period_toy("R1.pj", 1), 0, 4000)  # every attempt new: reached at attempt 4,000 well inside 161,000
    assert (just.count, just.attempts) == (4000, 4000)
    dup = fresh_toy("R1.dup", 0.99)
    d = cap.count(dup, 1, 200)
    assert (d.count, d.ended, d.attempts) == reference(dup, 1, 200) and d.count < 200 and d.ended == "budget"
    for toy in (period_toy("R1.pr1", 51), period_toy("R1.pr2", 52), period_toy("R1.pr3", 7)):
        for t in (3, 50, 100):
            assert (cap.count(toy, 0, t).count, cap.count(toy, 0, t).ended, cap.count(toy, 0, t).attempts) == reference(toy, 0, t)

def test_solver_agreement_and_every_item_built_is_checked() -> None:
    toy = rand_toy("R1.sv", 40)
    ok = cap.count(toy, 0, 30, solver=agree_solver)
    assert ok.n_disagreements == 0 and ok.first_disagreement_attempt is None and ok.solver_checked

    def odd(question: str, structured: Any) -> str:
        return str(structured["i"] + (1 if structured["i"] % 5 == 3 else 0))

    d = cap.count(toy, 0, 30, solver=odd)
    rng = stream(400002, "capacity", "R1", "R1.sv", "0")
    positions, items = [], []
    for n in range(1, d.attempts + 1):  # replay: every attempt, duplicates included, with item = 3 mod 5
        it = build_item(toy, 0, rng, "-", "x")
        if it.structured["i"] % 5 == 3:
            positions.append(n)
            items.append(it.structured["i"])
    assert d.n_disagreements == len(positions) and d.first_disagreement_attempt == positions[0]
    assert len(items) > len(set(items))  # a discarded duplicate of such an item was checked too
    assert d.disagreements[0].reason == "mismatch" and d.disagreements[0].attempt == positions[0]


def test_solver_error_nonstr_and_timeout_are_disagreements(monkeypatch: pytest.MonkeyPatch) -> None:
    toy = rand_toy("R1.sv2", 10**6)
    assert cap.count(toy, 0, 3, solver=lambda q, s: 1 / 0).disagreements[0].reason == "error:ZeroDivisionError"
    assert cap.count(toy, 0, 3, solver=lambda q, s: 5).disagreements[0].reason == "non-str"

def _traces(toy: Lineage, level: int, target: int) -> cap._Trace:
    tr = cap._Trace(toy.id, level)
    cap._advance(toy, level, target, tr, set(), stream(400002, *cap.stream_labels(toy, level)), None, dk.item_key)

# ------------------------------------------------------------- cells in processes: no dependence on workers


def _cells() -> list[tuple[str, int, int]]:
    return [(x, v, t) for x in ("R1.u", "R1.v", "R1.w") for v in (0, 3, 6) for t in (30, 90)]


def _source() -> ToySource:
    lins = {"R1.u": rand_toy("R1.u", 70), "R1.v": rand_toy("R1.v", 10**6), "R1.w": fresh_toy("R1.w", 0.9)}
    return ToySource(lins, {k: agree_solver for k in lins})


def _fingerprint(draws: list[cap.Draw]) -> list[Any]:
    return [(d.lineage, d.level, d.target, d.count, d.ended, d.attempts, d.keys, d.n_disagreements) for d in draws]


@pytest.mark.skipif(sys.platform == "win32", reason="fork")
def test_results_do_not_depend_on_the_number_of_workers_or_the_order(tmp_path: Path) -> None:
    tasks = _cells()
    one = cap.CellRunner(_source(), workers=1)
    two = cap.CellRunner(_source(), workers=3, mp_context="fork", cache_dir=tmp_path / "c", cache_tag="w")
    try:
        a = one.run(tasks)
        b = two.run(tasks)
        shuffled = tasks[:]
        random.Random(5).shuffle(shuffled)
        c = two.run(shuffled)
    finally:
        two.close()
    assert _fingerprint(a) == _fingerprint(b)
    by = dict(zip(shuffled, _fingerprint(c), strict=True))
    assert [by[t] for t in tasks] == _fingerprint(a)

def inputs(clusters: dict[str, list[str]], part_a: set[str], candidates: dict[str, list[str]]) -> tuple[dict, dict, dict]:
    cand = {"candidates": candidates}
    part = {"cluster_part": {c: ("A" if c in part_a else "B") for c in clusters}}
    mp = {"clusters": [{"cluster_id": c, "members": m} for c, m in sorted(clusters.items())]}
    return cand, part, mp


class Model:
    """The drafter's model of the draws: a lineage yields ``capacity`` keys at any level, so c(x, v, T) = min(capacity, T).

    ``shared`` gives pairs of lineages the number of keys the two key sets have in common; ``disagree`` names lineages whose
    solvers disagree (level -> first attempt); ``ended`` names the reason a short draw ended.
    """

    def __init__(self, capacity: dict[str, Any], shared: dict[tuple[str, str], int] | None = None,
                 disagree: dict[str, dict[int, int]] | None = None, ended: dict[str, str] | None = None) -> None:
        self.capacity, self.shared, self.disagree, self.ended = capacity, shared or {}, disagree or {}, ended or {}
        self.calls: list[tuple[str, int, int]] = []

    def cap(self, x: str, v: int) -> int:
        c = self.capacity[x]
        return c[v] if isinstance(c, dict) else c

    def __call__(self, x: str, v: int, t: int) -> cap.Draw:
        self.calls.append((x, v, t))
        n = min(self.cap(x, v), t)
        base = (sorted(self.capacity).index(x) + 1) << 40
        common = sum(m for (a, b), m in self.shared.items() if x in (a, b))
        keys = []
        for i in range(n):
            if i < common:
                pair = next(p for p in self.shared if x in p)
                keys.append(((9 << 60) + (sorted(self.capacity).index(pair[0]) << 20) + i).to_bytes(16, "big"))
            else:
                keys.append((base + i).to_bytes(16, "big"))
        dis = self.disagree.get(x, {}).get(v)
        return cap.Draw(x, v, t, n, "target" if n >= t else self.ended.get(x, "budget"), t if n >= t else 40 * t + 1000,
                        b"".join(keys), True, 0 if dis is None else 3, (), dis)


def run_model(model: Model, clusters: dict[str, list[str]], part_a: set[str], candidates: dict[str, list[str]],
              levels: tuple[int, ...] = (0,)) -> dict[str, Any]:
    cand, part, mp = inputs(clusters, part_a, candidates)
    return cap.run_stage4(cand, part, mp, None, levels=levels, count_fn=model)


def trail(part: dict[str, Any]) -> list[tuple[int, list[str]]]:
    out = []
    for r in part["rounds"]:
        union = sorted(x for c, cr in r["clusters"].items() if cr["union_evaluated"] and any(
            n < r["D_G"] for n in cr["union_by_level"].values()) for x in r["clusters"][c]["lineages"])
        out.append((r["G"], sorted(set(r["failing"]) - set(union))))
    return out


def test_worked_example_4a_of_the_drafter() -> None:

def test_names_relabelled_is_answer_preserving_on_real_lineages(monkeypatch: pytest.MonkeyPatch) -> None:
    """Rename every name one-to-one in the question, the structured form and the key answer: the key must not change
    and the second solver must return the renamed answer. A key that treated a data word as a name would fail here."""
    monkeypatch.setattr(cap, "LABEL", "capacity-smoke")
    import re

    try:
        src = _real()
        picks = ["M1.shopping_r3", "R1.knights_knaves_r6", "R1.temporal_weekday_shift_r6", "R2.cheapest_network", "R1.categories_default_exceptions_r4"]
        lins = [src.lineage(x) for x in picks]
    except ImportError:  # pragma: no cover
        pytest.skip("generators not importable")
    classes, _ = dk._vocabulary()
    for toy in lins:
        rng = stream(3, "capacity-smoke-names", toy.id)
        for level in (0, 3, 6):
            it = build_item(toy, level, rng, "-", "x")
            sj = json.dumps(it.structured)
            cores: dict[str, str] = {}
            for txt in (it.question, sj, it.answer):
                for _s, _e, core, _suf in dk.find_names(txt):
                    cores.setdefault(core.casefold(), core)
            used = {w.casefold() for w in re.findall(r"[A-Za-z][A-Za-z'-]*", it.question + sj + it.answer)}
            by_class: dict[str, list[str]] = {}
            for k, core in cores.items():
                c = next(c for c in dk.NAME_CLASSES if core in classes[c] or core[0].lower() + core[1:] in classes[c])
                by_class.setdefault(c, []).append(k)
            mp: dict[str, str] = {}
            for c, ks in by_class.items():
                if len(ks) >= 2:
                    mp.update({k: ks[(i + 1) % len(ks)] for i, k in enumerate(ks)})
                else:
                    mp[ks[0]] = next(w for w in sorted(classes[c]) if w.casefold() not in used).casefold()
            spelling = {w.casefold(): w for c in dk.NAME_CLASSES for w in classes[c]}

            def rename(text: str, mp: dict[str, str] = mp, spelling: dict[str, str] = spelling) -> str:
                out, at = [], 0
                for s, e, core, suf in dk.find_names(text):
                    new = spelling[mp[core.casefold()]]
                    if new[0].islower() and core[0].isupper():
                        new = new[0].upper() + new[1:]
                    out += [text[at:s], new + suf]
                    at = e + len(suf)
                out.append(text[at:])
                return "".join(out)

            st2 = json.loads(rename(sj))
            assert dk.item_key(toy.id, rename(it.question), st2) == dk.item_key(toy.id, it.question, json.loads(sj)), toy.id
            assert src.solver(toy.id)(rename(it.question), st2) == rename(it.answer), toy.id


def test_prefix_property_and_cache_continuation_on_real_lineages(monkeypatch: pytest.MonkeyPatch, tmp_path: Path) -> None:
    """On three real generators (label ``capacity-smoke``, targets of a few hundred): the draw for a lower target is a prefix of
    the draw for a higher one, a cache answers it, and a cache continues the stream to the same draw as one made from the start."""
    monkeypatch.setattr(cap, "LABEL", "capacity-smoke")
    try:
        src = _real()
        picks = [src.lineage(x) for x in ("R1.compass_turns", "M1.tiered_pricing", "C1.postfix_evaluation")]
    except ImportError:  # pragma: no cover
        pytest.skip("generators not importable")
    for toy in picks:
        for level in (0, 6):
            cache = cap.TraceCache(tmp_path / f"{toy.id}-{level}", "real")
            low = cap.count(toy, level, 120, cache=cache)
            high = cap.count(toy, level, 400, cache=cache)  # continued from the stored state
            fresh_high = cap.count(toy, level, 400)
            assert (high.count, high.ended, high.attempts, high.keys) == (fresh_high.count, fresh_high.ended, fresh_high.attempts, fresh_high.keys)
            assert high.keys[: 16 * low.count] == low.keys  # the lower target is a prefix of the higher
            again = cap.count(toy, level, 120, cache=cache)  # answered from the longer trace
            assert (again.count, again.attempts, again.keys) == (low.count, low.attempts, low.keys)
            assert (low.count, low.ended) == (120, "target")

                "--workers", str(w), "--mp-context", "fork", "--cache-dir", str(tmp_path / f"cache{w}"), "--cache-tag", "cli"]
        assert cap.main(argv) == 0
        assert cap.main(argv) == 2  # an existing output is never overwritten
        outs.append(json.loads(out.read_text()))
    for o in outs:
        o.pop("inputs")
    assert cap.dumps(outs[0]) == cap.dumps(outs[1])
    a = outs[0]["parts"]
    assert a["R1"]["A"]["G_registered"] == 1 and a["M1"]["A"]["G_registered"] == 1 and a["C1"]["A"]["G_registered"] == 1
    assert a["R1"]["A"]["rounds"][0]["D_G"] == cap.demand(1) == 10 * (4 + 20 + 1)
    assert outs[0]["spec"]["solvers_checked"] is True and outs[0]["spec"]["reduced_levels"] is False


def test_a_draw_that_gave_up_is_cached_and_not_redrawn(tmp_path: Path) -> None:
    calls = {"n": 0}
    base = give_up_toy("R1.gq", 4)

    def counted(level: int, rng: random.Random) -> Built:
        calls["n"] += 1
        return base.build(level, rng)

    toy = lin("R1.gq", counted)
    cache = cap.TraceCache(tmp_path / "c", "t")
    first = cap.count(toy, 0, 50, cache=cache)
    built = calls["n"]
    second = cap.count(toy, 0, 90, cache=cache)
    assert (first.count, first.ended) == (4, "gave_up") == (second.count, second.ended) and calls["n"] == built
    assert any(p.name.endswith("g.trace") for p in (tmp_path / "c").iterdir())
